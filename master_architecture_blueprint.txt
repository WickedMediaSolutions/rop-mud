### 1.3 GATING ENGINE — The Central Permission Check

```python
# world/race_class_matrix.py (continued)

def can_cast_spells(character) -> bool:
    """Master gate: does this character's class allow ANY spellcasting?"""
    char_class = character.attributes.get("class", default="Warrior")
    schools = CLASS_SPELL_SCHOOLS.get(char_class, [])
    return len(schools) > 0

def can_learn_spell(character, spell_key: str) -> Tuple[bool, str]:
    """
    Check if a character's race/class combination permits learning a spell.
    Returns (allowed, reason).
    Called by: SpellHandler.can_cast(), _grant_spells_for_level(), guildmaster training.
    """
    char_class = character.attributes.get("class", default="Warrior")
    char_race = character.attributes.get("race", default="Human")
    char_level = character.attributes.get("level", default=1)

    spell = get_spell(spell_key)
    if not spell:
        return False, "Unknown spell."

    # 1. Class must allow this spell's school
    allowed_schools = CLASS_SPELL_SCHOOLS.get(char_class, [])
    if spell["school"] not in allowed_schools:
        return False, f"{char_class}s cannot cast {spell['school']} spells."

    # 2. Class must allow spells up to this spell's level
    max_lvl_map = CLASS_MAX_SPELL_LEVEL.get(char_class, {})
    max_for_school = max_lvl_map.get(spell["school"], 80)
    if spell["level"] > max_for_school:
        return False, f"{char_class}s can only learn {spell['school']} spells up to level {max_for_school}."

    # 3. Race-level restrictions (e.g., Orc can't learn any spells)
    race_data = RACES.get(char_race, {})
    race_max = race_data.get("max_spell_level_by_school", {}).get(spell["school"], 80)
    if spell["level"] > race_max:
        return False, f"{char_race}s cannot learn {spell['school']} spells above level {race_max}."

    # 4. Character level check
    if spell["level"] > char_level:
        return False, f"You must be level {spell['level']} to learn {spell['name']}."

    return True, ""

def can_use_skill(character, skill_key: str) -> Tuple[bool, str]:
    """Check if a character's class permits using a physical combat skill."""
    char_class = character.attributes.get("class", default="Warrior")
    class_data = CLASSES.get(char_class, {})
    class_skills = class_data.get("class_skills", [])
    if skill_key not in class_skills:
        return False, f"{char_class}s cannot use {skill_key}."
    return True, ""

def can_equip_slot(character, slot: str, item_type: str) -> Tuple[bool, str]:
    """Check race/class equipment restrictions."""
    char_race = character.attributes.get("race", default="Human")
    char_class = character.attributes.get("class", default="Warrior")
    race_data = RACES.get(char_race, {})
    class_data = CLASSES.get(char_class, {})

    # Race restrictions (e.g., Pixie can't wear heavy armor)
    if slot in race_data.get("forbidden_slots", []):
        return False, f"{char_race}s cannot equip items in the {slot} slot."

    # Class armor proficiency
    if item_type.startswith("armor_"):
        armor_type = item_type.replace("armor_", "")
        if armor_type not in class_data.get("allowed_armor_types", []):
            return False, f"{char_class}s cannot wear {armor_type} armor."

    # Class weapon proficiency
    if item_type.startswith("weapon_"):
        weapon_type = item_type.replace("weapon_", "")
        if weapon_type not in class_data.get("allowed_weapon_types", []):
            return False, f"{char_class}s cannot wield {weapon_type} weapons."

    return True, ""
```

```python
@dataclass
class PracticeSession:
    """Tracks a player's practice points and trained skills."""
    practice_points: int = 0          # Earned on level-up, spent at guildmasters
    trained_spells: Set[str] = field(default_factory=set)   # Spells unlocked via training

def award_practice_points(character, level: int):
    """Called on level-up. Awards practice points based on class."""
    char_class = character.attributes.get("class", default="Warrior")
    # Casters get more points (need them for spells), martials get fewer
    pp_per_level = {
        "Warrior": 3, "Paladin": 4, "Cleric": 5, "Mage": 6,
        "Rogue": 3, "Warlock": 5, "Druid": 5, "Ranger": 4,
        "Monk": 3, "Necromancer": 5,
    }
    points = pp_per_level.get(char_class, 3)
    session = character.attributes.get("practice_session", default=None)
    if session is None:
        session = PracticeSession()
    session.practice_points += points
    character.attributes.add("practice_session", session)
    character.msg(f"|gYou have earned {points} practice points! (Total: {session.practice_points})|n")


# Guildmaster NPC typeclass — world/guildmaster_npc.py

class GuildmasterNPC(DefaultCharacter):
    """
    NPC that allows players to spend practice points to learn spells/skills.
    Only shows spells/skills valid for the player's race/class combination.
    """
    def at_object_creation(self):
        super().at_object_creation()
        self.db.guild_class = "Warrior"  # Which class this guildmaster serves

    def get_trainable_spells(self, character) -> List[Dict]:
        """Return spells this character can train, with practice point costs."""
        from world.race_class_matrix import can_learn_spell
        from world.spells import SPELLS
        char_level = character.attributes.get("level", default=1)
        learned = set(character.attributes.get("learned_spells", default=[]))
        session = character.attributes.get("practice_session", PracticeSession())
        trained = session.trained_spells

        available = []
        for key, spell in SPELLS.items():
            if spell["name"] in learned or spell["name"] in trained:
                continue
            if spell["level"] > char_level:
                continue
            allowed, _ = can_learn_spell(character, key)
            if not allowed:
                continue
            cost = 2  # Base practice point cost per spell
            available.append({"name": spell["name"], "level": spell["level"],
                              "school": spell["school"], "cost": cost})
        return sorted(available, key=lambda s: s["level"])

    def train_spell(self, character, spell_name: str) -> Tuple[bool, str]:
        """Spend practice points to learn a spell."""
        session = character.attributes.get("practice_session", PracticeSession())
        trainable = {s["name"]: s for s in self.get_trainable_spells(character)}
        if spell_name not in trainable:
            return False, "You cannot train that spell here."
        info = trainable[spell_name]
        if session.practice_points < info["cost"]:
            return False, f"You need {info['cost']} practice points (have {session.practice_points})."
        session.practice_points -= info["cost"]
        session.trained_spells.add(spell_name)
        # Add to learned spells
        learned = character.attributes.get("learned_spells", default=[])
        learned.append(spell_name)
        character.attributes.add("learned_spells", learned)
        character.attributes.add("practice_session", session)
        return True, f"You learn {spell_name}! ({session.practice_points} practice points remaining)"
```

```python
# world/combat_state.py — Full combat state machine

from enum import Enum, auto

class CombatState(Enum):
    IDLE = auto()           # Not in combat
    ENGAGING = auto()       # First round, initiating attack
    FIGHTING = auto()       # Active auto-attack rounds
    FLEEING = auto()        # Attempting to flee (one-round delay)
    STUNNED = auto()        # Cannot act this round
    UNCONSCIOUS = auto()    # HP = 0, pending death/rescue
    DEAD = auto()           # Defeated, awaiting respawn/corpse

class CombatStateMachine:
    """Per-character combat state with transition validation."""

    VALID_TRANSITIONS = {
        CombatState.IDLE:         [CombatState.ENGAGING],
        CombatState.ENGAGING:     [CombatState.FIGHTING, CombatState.FLEEING, CombatState.IDLE],
        CombatState.FIGHTING:     [CombatState.FLEEING, CombatState.STUNNED, CombatState.UNCONSCIOUS, CombatState.IDLE],
        CombatState.FLEEING:      [CombatState.IDLE, CombatState.FIGHTING],  # Success → IDLE, Fail → FIGHTING
        CombatState.STUNNED:      [CombatState.FIGHTING, CombatState.UNCONSCIOUS],
        CombatState.UNCONSCIOUS:  [CombatState.DEAD, CombatState.IDLE],  # IDLE = revived
        CombatState.DEAD:         [CombatState.IDLE],  # After respawn
    }

    @staticmethod
    def transition(character, new_state: CombatState) -> bool:
        current = character.ndb.combat_state or CombatState.IDLE
        if new_state not in CombatStateMachine.VALID_TRANSITIONS.get(current, []):
            return False
        character.ndb.combat_state = new_state
        return True
```

```python
# world/damage_formulas.py

import random
from enum import Enum

class DamageType(Enum):
    SLASH = "slash"
    PIERCE = "pierce"
    BLUNT = "blunt"
    MAGIC_FIRE = "magic_fire"
    MAGIC_COLD = "magic_cold"
    MAGIC_LIGHTNING = "magic_lightning"
    MAGIC_SHADOW = "magic_shadow"
    MAGIC_HOLY = "magic_holy"
    POISON = "poison"
    BLEED = "bleed"

# Armor mitigation per damage type (percentage reduction)
ARMOR_MITIGATION = {
    DamageType.SLASH:   0.15,   # Chain/plate good vs slashing
    DamageType.PIERCE:  0.10,   # Harder to mitigate piercing
    DamageType.BLUNT:   0.05,   # Blunt crushes through armor
    DamageType.MAGIC_FIRE: 0.0, # Armor doesn't help vs magic
    DamageType.MAGIC_COLD: 0.0,
    DamageType.MAGIC_LIGHTNING: 0.0,
    DamageType.MAGIC_SHADOW: 0.0,
    DamageType.MAGIC_HOLY: 0.0,
    DamageType.POISON:  0.0,
    DamageType.BLEED:   0.0,
}

def calculate_melee_damage(attacker, defender, weapon_damage: int, damage_type: DamageType) -> dict:
    """
    Full melee damage calculation incorporating:
    - Base weapon damage
    - STR bonus (+1 per 2 STR above 10)
    - Critical hit chance (based on DEX, +5% per 5 DEX above 10)
    - Armor mitigation (based on defender's equipped armor + CON)
    - Damage type modifiers
    - Random variance (±20%)

    Returns: {"damage": int, "crit": bool, "absorbed": int, "type": DamageType}
    """
    stats_att = attacker.attributes.get("stats", {})
    stats_def = defender.attributes.get("stats", {})

    str_val = stats_att.get("str", 10)
    dex_val = stats_att.get("dex", 10)
    con_val = stats_def.get("con", 10)

    # Base damage
    str_bonus = max(0, (str_val - 10) // 2)
    base = weapon_damage + str_bonus

    # Critical hit check (DEX-based)
    crit_chance = max(0.02, (dex_val - 10) * 0.01)  # 1% per DEX above 10, min 2%
    is_crit = random.random() < crit_chance
    if is_crit:
        base = int(base * 1.5)  # 50% bonus on crit

    # Armor mitigation
    armor_value = _get_armor_value(defender)  # Sum of equipped armor pieces
    armor_mitigation_pct = ARMOR_MITIGATION.get(damage_type, 0.10)
    con_bonus = max(0, (con_val - 10) // 3)  # CON adds flat reduction
    absorbed = int(base * armor_mitigation_pct) + con_bonus
    absorbed = min(absorbed, base - 1)  # Always do at least 1 damage

    # Variance
    variance = random.uniform(0.80, 1.20)
    final_damage = max(1, int((base - absorbed) * variance))

    return {
        "damage": final_damage,
        "crit": is_crit,
        "absorbed": absorbed,
        "type": damage_type,
    }

def calculate_magic_damage(caster, target, spell_base: int, spell_per_level: int,
                           damage_type: DamageType) -> dict:
    """Magic damage: scales with INT/WIS, reduced by magic resistance."""
    stats_caster = caster.attributes.get("stats", {})
    stats_target = target.attributes.get("stats", {})

    casting_stat = max(stats_caster.get("int", 10), stats_caster.get("wis", 10))
    caster_level = caster.attributes.get("level", 1)

    base = spell_base + (spell_per_level * caster_level) + int(casting_stat * 0.6)

    # Magic resistance (based on target WIS + any resist gear)
    wis_resist = max(0, (stats_target.get("wis", 10) - 10) * 0.02)  # 2% per WIS above 10
    gear_resist = _get_magic_resist(target)  # From equipment
    total_resist = min(0.75, wis_resist + gear_resist)  # Cap at 75%

    final = max(1, int(base * (1.0 - total_resist)))
    return {"damage": final, "type": damage_type, "resisted_pct": total_resist}

def _get_armor_value(character) -> int:
    """Sum armor values from all equipped gear."""
    equipped = character.attributes.get("equipped", default={})
    total = 0
    for slot, item_name in equipped.items():
        for obj in character.contents:
            if obj.key == item_name:
                total += obj.attributes.get("armor", default=0)
    # Natural armor from race (e.g., Lizardfolk +4, Mountain Dwarf +5)
    race_data = RACES.get(character.attributes.get("race", ""), {})
    total += race_data.get("natural_armor", 0)
    return total

def _get_magic_resist(character) -> float:
    """Sum magic resistance from gear and racial bonuses."""
    equipped = character.attributes.get("equipped", default={})
    total = 0.0
    for slot, item_name in equipped.items():
        for obj in character.contents:
            if obj.key == item_name:
                total += obj.attributes.get("magic_resist", default=0.0)
    return total
```

```python
            skill_name = queued_skill
            character.ndb.queued_skill = None
            self._execute_skill_attack(character, target, skill_name)
        else:
            # Normal auto-attack
            CombatHandler._execute_attack_round(character, target)

        # Check if a combat spell is being cast (spells resolve instantly,
        # but the ticker checks if the caster is stunned/silenced)
        if character.ndb.casting_spell:
            # Spell casting is handled by SpellHandler.cast() directly,
            # but we validate the caster isn't stunned
            if character.attributes.get("stunned", False):
                character.msg("|rYou are stunned and cannot cast!|n")
                character.ndb.casting_spell = None

    def _execute_skill_attack(self, character, target, skill_name):
        """Execute a special physical skill within the combat round."""
        skill = COMBAT_SKILLS.get(skill_name)
        if not skill:
            return

        # Check stamina
        stamina = character.attributes.get("stamina", default=100)
        if stamina < skill["stamina_cost"]:
            character.msg(f"|yNot enough stamina for {skill['name']}.|n")
            return

        # Deduct stamina
        character.attributes.add("stamina", stamina - skill["stamina_cost"])

        # Hit roll (same as normal attack)
        if not _roll_attack_hit(character, target):
            character.msg(f"Your {skill['name']} misses {target.key}!")
            return

        # Damage calculation with skill multiplier
        weapon_dmg = _get_weapon_damage(character)
        result = calculate_melee_damage(character, target, weapon_dmg, skill["damage_type"])
        result["damage"] = int(result["damage"] * skill["damage_mult"])

        _apply_damage_to_target(character, target, result["damage"])

        # Stun check
        if random.random() < skill.get("stun_chance", 0):
            target.attributes.add("stunned", True)
            target.msg("|mYou have been stunned!|n")
            from evennia.utils import delay
            delay(3.0, lambda t=target: t.attributes.add("stunned", False))

        # Set cooldown
        cooldowns = character.attributes.get("skill_cooldowns", {})
        cooldowns[skill_name] = time.time() + skill["cooldown"]
        character.attributes.add("skill_cooldowns", cooldowns)
```

```python
    ai = mob.attributes.get("mob_ai")
    if not ai:

    if ai.disposition == MobDisposition.PASSIVE:

        return False  # Player too low level, not worth aggro

    player_align = player.attributes.get("alignment", "")
    mob_align = mob.attributes.get("alignment", "")
    if player_align == mob_align and mob_align:
        return False  # Same faction, no aggro
```

```python
# world/garbage_collection.py — Cleanup ticker for corpses, orphaned mobs, expired effects

class GarbageCollectionScript(DefaultScript):
    """
    Global cleanup script. Runs every 60 seconds.
    Handles:
      - Corpse decay (delete corpses older than CORPSE_DECAY_SECONDS)
      - Orphaned mob cleanup (mobs with no players in zone for > 30 min)
      - Expired buff/debuff removal
      - Stale combat state cleanup
    """

    CORPSE_DECAY_SECONDS = 600      # 10 minutes
    ORPHANED_MOB_TIMEOUT = 1800     # 30 minutes
    STALE_COMBAT_TIMEOUT = 300      # 5 minutes

    def at_script_creation(self):
        self.key = "garbage_collector"
        self.interval = 60
        self.persistent = True

    def at_repeat(self):
        now = time.time()

        # 1. Decay expired corpses
        for obj in ObjectDB.objects.all():
            if obj.attributes.get("is_corpse", False):
                created = obj.attributes.get("corpse_created_at", now)
                if now - created > self.CORPSE_DECAY_SECONDS:
                    # Move any remaining items to the room before deleting
                    location = obj.location
                    for item in obj.contents:
                        if not item.destination:
                            item.move_to(location, quiet=True)
                    if location:
                        location.msg_contents(f"|y{obj.key} crumbles to dust.|n")
                    obj.delete()

        # 2. Despawn orphaned mobs (no players in zone)
        #    Track last_player_seen timestamp on rooms
        for obj in ObjectDB.objects.all():
            if obj.attributes.get("is_mob", False) and not obj.has_account:
                location = obj.location
                if location:
                    last_seen = location.attributes.get("last_player_seen", now)
                    if now - last_seen > self.ORPHANED_MOB_TIMEOUT:
                        # Despawn this mob (it'll respawn via spawner)
                        obj.delete()

        # 3. Clean up stale combat ndb references
        for char in ObjectDB.objects.all():
            if not hasattr(char, "ndb"):
                continue
            if char.ndb.in_combat:
                target = char.ndb.combat_target
                if target and not _is_valid_target(char, target):
                    CombatHandler.stop_combat(char)

        # 4. Remove expired buffs/debuffs
        for char in ObjectDB.objects.all():
            if not hasattr(char, "attributes"):
                continue
            # Check stunned flag — if set but no delay pending, clear it
            # (handled by delay() callbacks, but this is a safety net)
            buffs = char.attributes.get("active_buffs", default=[])
            active = []
            for buff in buffs:
                if buff.get("expires_at", now + 1) > now:
                    active.append(buff)
                else:
                    # Buff expired — revert its effects
                    self._revert_buff(char, buff)
            char.attributes.add("active_buffs", active)

    def _revert_buff(self, char, buff):
        """Revert stat changes from an expired buff."""
        if buff["type"] == "stat_boost":
            stats = char.attributes.get("stats", {})
            stat = buff["stat"]
            stats[stat] = max(1, stats.get(stat, 10) - buff["amount"])
            char.attributes.add("stats", stats)
```

```python
def build():
    """Execute the full zone build."""
    created_rooms = {}

    # 1. Create rooms
    for room_key, room_data in ROOMS.items():
        room = create_object("typeclasses.rooms.Room", key=room_data["key"])
        room.db.desc = room_data["desc"]
        for attr_key, attr_val in room_data.get("attributes", []):
            room.attributes.add(attr_key, attr_val)
        for tag_name, tag_category in room_data.get("tags", []):
            room.tags.add(tag_name, category=tag_category)
        created_rooms[room_key] = room
        print(f"  Created room: {room.key}")

    # 2. Create exits
    for src_key, exit_name, dst_key, aliases in EXITS:
        src_room = created_rooms[src_key]
        dst_room = created_rooms[dst_key]
        exit_obj = create_object(
            "typeclasses.exits.Exit",
            key=exit_name,
            location=src_room,
            destination=dst_room,
        )
        exit_obj.aliases.add(aliases)
        print(f"  Created exit: {src_room.key} -> {exit_name} -> {dst_room.key}")

    # 3. Create mob spawners
    for spawner_key, spawner_data in MOB_SPAWNERS.items():
        room = created_rooms[spawner_data["room_key"]]
        spawner = create_object("typeclasses.objects.MobSpawner", key=spawner_key, location=room)
        spawner.attributes.add("mob_prototype", spawner_data["mob_prototype"])
        spawner.attributes.add("max_spawned", spawner_data["max_spawned"])
        spawner.attributes.add("respawn_delay", spawner_data["respawn_delay"])
        spawner.attributes.add("spawn_message", spawner_data["spawn_message"])
        print(f"  Created spawner: {spawner_key} in {room.key}")

    # 4. Create shopkeepers
    for shop_key, shop_data in SHOPKEEPERS.items():
        room = created_rooms[shop_data["room_key"]]
        shopkeeper = create_object(shop_data["typeclass"], key=shop_data["key"], location=room)
        shopkeeper.db.desc = shop_data["desc"]
        for attr_key, attr_val in shop_data.get("attributes", []):
            shopkeeper.attributes.add(attr_key, attr_val)
        print(f"  Created shopkeeper: {shopkeeper.key} in {room.key}")

    # 5. Create guildmasters
    for gm_key, gm_data in GUILDMASTERS.items():
        room = created_rooms[gm_data["room_key"]]
        gm = create_object(gm_data["typeclass"], key=gm_data["key"], location=room)
        gm.db.desc = gm_data["desc"]
        for attr_key, attr_val in gm_data.get("attributes", []):
            gm.attributes.add(attr_key, attr_val)
        print(f"  Created guildmaster: {gm.key} in {room.key}")

    print(f"\n|gZone '{HEADER['zone_name']}' built successfully!|n")
    print(f"  Rooms: {len(created_rooms)}")
    print(f"  Exits: {len(EXITS)}")
    print(f"  Spawners: {len(MOB_SPAWNERS)}")
    print(f"  Shopkeepers: {len(SHOPKEEPERS)}")
    print(f"  Guildmasters: {len(GUILDMASTERS)}")
```

### 5.2 MOB SPAWNER TYPECLASS

```python
# typeclasses/objects.py — MobSpawner object

class MobSpawner(DefaultObject):
    """
    Room-level spawner that creates and manages mob instances.
    Attached to a room. Periodically checks if mobs need respawning.
    """

    def at_object_creation(self):
        super().at_object_creation()
        self.db.mob_prototype = ""       # Key in MOB_PROTOTYPES
        self.db.max_spawned = 1
        self.db.respawn_delay = 120      # Seconds
        self.db.spawn_message = "A creature appears."
        self.db.current_spawned = []     # List of spawned mob dbrefs

    def at_init(self):
        """Start the spawn check loop."""
        self._start_spawn_ticker()

    def _start_spawn_ticker(self):
        """Check every 30 seconds if we need to spawn."""
        from evennia.scripts.scripts import DefaultScript

        class SpawnCheckScript(DefaultScript):
            def at_script_creation(self):
                self.key = f"spawn_check_{self.obj.id}"
                self.interval = 30
                self.persistent = False

            def at_repeat(self):
                spawner = self.obj
                # Clean up dead mobs from tracking list
                alive = []
                for mob_dbref in spawner.db.current_spawned:
                    mob = ObjectDB.objects.filter(id=mob_dbref).first()
                    if mob and mob.attributes.get("hp", 0) > 0:
                        alive.append(mob_dbref)
                spawner.db.current_spawned = alive

                # Spawn if below max
                if len(alive) < spawner.db.max_spawned:
                    spawner._spawn_mob()

        self.scripts.add(SpawnCheckScript)

    def _spawn_mob(self):
        """Create one mob from the prototype definition."""
        from world.batch_zones.example_zone import MOB_PROTOTYPES
        proto = MOB_PROTOTYPES.get(self.db.mob_prototype)
        if not proto:
            return

        mob = create_object(proto["typeclass"], key=proto["key"], location=self.location)
        for attr_key, attr_val in proto.get("attributes", []):
            mob.attributes.add(attr_key, attr_val)

        # Store loot table on mob for corpse creation
        mob.attributes.add("loot_table", proto.get("loot_table", []))

        self.db.current_spawned.append(mob.id)

        if self.db.spawn_message:
            self.location.msg_contents(f"|g{self.db.spawn_message}|n")
```

## APPENDIX C: CRITICAL BUG FIX — ORC WARRIOR SPELL ACCESS

**Root Cause:** Three missing gates:
1. `CmdCast` and `CmdSpells` are added to ALL CharacterCmdSets unconditionally
2. `SpellHandler.can_cast()` only checks level/mana/cooldown, not race/class
3. `_grant_spells_for_level()` grants all level-appropriate spells to all characters

**Fix (3-line summary):**
1. Add `locks = "cmd:can_cast_spells()"` to CmdCast and CmdSpells
2. Add `can_learn_spell()` check at the top of `SpellHandler.can_cast()`
3. Add `can_learn_spell()` filter in `_grant_spells_for_level()`
